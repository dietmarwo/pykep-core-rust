cmake_minimum_required(VERSION 3.28)
project(pykep_matched_benchmarks LANGUAGES CXX)
find_package(kep3 3.1 CONFIG REQUIRED)
add_executable(pykep-native-benchmark benchmark.cpp)
add_executable(pykep-export-fixtures export.cpp)
target_compile_definitions(pykep-export-fixtures PRIVATE PYKEP_WITH_SDK)
foreach(target pykep-native-benchmark pykep-export-fixtures)
    target_compile_features(${target} PRIVATE cxx_std_23)
    target_link_libraries(${target} PRIVATE kep3::kep3)
endforeach()
